/* ════════════════════════════════════════
   상수 & 초기 데이터
════════════════════════════════════════ */
const TIERS        = ['S', 'A', 'B', 'C', 'unranked'];
// 클릭 순환 순서
const CYCLE        = ['unranked', 'S', 'A', 'B', 'C'];
const STORAGE_KEY  = 'movie-tierlist-v2';

// 기획서에 명시된 초기 영화 목록 (최초 실행 시에만 사용)
const DEFAULT_MOVIES = [
  '기생충', '인터스텔라', '라라랜드',
  '센과 치히로의 행방불명', '어벤져스: 엔드게임',
  '범죄도시', '타이타닉', '토이 스토리'
];

/* ════════════════════════════════════════
   상태 (State)
   { id: string, title: string, tier: string }[]
════════════════════════════════════════ */
let movies = [];

/* ════════════════════════════════════════
   LocalStorage 유틸
════════════════════════════════════════ */
function saveToStorage() {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(movies));
}

function loadFromStorage() {
  const raw = localStorage.getItem(STORAGE_KEY);
  if (raw) {
    try { movies = JSON.parse(raw); } catch { movies = []; }
  } else {
    // 최초 실행: 기본 영화를 미분류로 추가
    movies = DEFAULT_MOVIES.map((title, i) => ({
      id: `m_${Date.now()}_${i}`,
      title,
      tier: 'unranked'
    }));
    saveToStorage();
  }
}

/* ════════════════════════════════════════
   렌더링
════════════════════════════════════════ */
function render() {
  // 각 티어 컨테이너 초기화
  TIERS.forEach(tier => {
    document.getElementById(`tier-${tier}`).innerHTML = '';
  });

  // 영화 카드 생성 및 배치
  movies.forEach(movie => {
    const card = createCard(movie);
    document.getElementById(`tier-${movie.tier}`).appendChild(card);
  });
}

function createCard(movie) {
  const card = document.createElement('div');
  card.className   = 'movie-card';
  card.textContent = movie.title;
  card.title       = movie.title;   // 브라우저 기본 툴팁
  card.dataset.id  = movie.id;

  // ── 드래그 앤 드롭 ──
  card.draggable = true;
  card.addEventListener('dragstart', onDragStart);
  card.addEventListener('dragend',   onDragEnd);

  // ── 클릭 순환 이동 ──
  card.addEventListener('click', onCardClick);

  return card;
}

/* ════════════════════════════════════════
   영화 추가
════════════════════════════════════════ */
function addMovie(title) {
  title = title.trim();
  if (!title) return;

  // 중복 체크 (같은 제목이 이미 있으면 경고)
  if (movies.some(m => m.title === title)) {
    showToast('⚠️ 이미 추가된 영화입니다.', '#f59e0b');
    return;
  }

  const newMovie = {
    id:    `m_${Date.now()}`,
    title,
    tier:  'unranked'
  };
  movies.push(newMovie);
  saveToStorage();
  render();
  showToast(`✅ "${title}" 추가 완료!`);
}

/* ════════════════════════════════════════
   클릭 순환 이동
════════════════════════════════════════ */
function onCardClick(e) {
  // 드래그 직후 click 이벤트가 발생하는 것을 방지
  if (isDragging) return;

  const id    = e.currentTarget.dataset.id;
  const movie = movies.find(m => m.id === id);
  if (!movie) return;

  const currentIdx = CYCLE.indexOf(movie.tier);
  movie.tier = CYCLE[(currentIdx + 1) % CYCLE.length];

  saveToStorage();
  render();
}

/* ════════════════════════════════════════
   드래그 앤 드롭
════════════════════════════════════════ */
let dragId    = null;   // 현재 드래그 중인 카드 id
let isDragging = false;

function onDragStart(e) {
  isDragging = true;
  dragId     = e.currentTarget.dataset.id;
  e.currentTarget.classList.add('dragging');
  // Firefox 호환: dataTransfer 필수
  e.dataTransfer.effectAllowed = 'move';
  e.dataTransfer.setData('text/plain', dragId);
}

function onDragEnd(e) {
  e.currentTarget.classList.remove('dragging');
  // 약간의 딜레이 후 isDragging 해제 (click 이벤트 방지)
  setTimeout(() => { isDragging = false; }, 100

AI: # 드래그 앤 드롭 \~ 끝 부분 (이어서)

`onDragEnd` 의 `setTimeout` 닫는 부분부터 파일 끝까지입니다. 이전 코드에 그대로 이어 붙이세요!

  // (이전 코드에서 이어짐)

/* 티어 행(row)에 dragover / drop 이벤트 등록 */
function setupDropZones() {
  document.querySelectorAll('.tier-row').forEach(row => {
    // dragover: 드롭 허용 신호
    row.addEventListener('dragover', e => {
      e.preventDefault();
      e.dataTransfer.dropEffect = 'move';
      row.classList.add('drag-over');
    });

    // dragleave: 강조 제거
    row.addEventListener('dragleave', e => {
      // 자식 요소로 이동할 때 오발 방지
      if (!row.contains(e.relatedTarget)) {
        row.classList.remove('drag-over');
      }
    });

    // drop: 실제 이동 처리
    row.addEventListener('drop', e => {
      e.preventDefault();
      row.classList.remove('drag-over');

      const targetTier = row.dataset.tier;
      if (!dragId || !targetTier) return;

      const movie = movies.find(m => m.id === dragId);
      if (!movie) return;

      // 같은 티어면 무시
      if (movie.tier === targetTier) return;

      movie.tier = targetTier;
      saveToStorage();
      render();
    });
  });
}

/* ════════════════════════════════════════
   JSON 저장 (다운로드)
════════════════════════════════════════ */
function saveAsJSON() {
  // 티어별로 그룹핑해서 보기 좋게 정리
  const output = {};
  TIERS.forEach(tier => {
    output[tier] = movies
      .filter(m => m.tier === tier)
      .map(m => m.title);
  });

  const blob = new Blob(
    [JSON.stringify(output, null, 2)],
    { type: 'application/json' }
  );
  const url  = URL.createObjectURL(blob);
  const link = document.createElement('a');
  link.href     = url;
  link.download = `tierlist_${getDateStamp()}.json`;
  link.click();
  URL.revokeObjectURL(url);   // 메모리 해제

  showToast('💾 JSON 파일 저장 완료!');
}

/* 파일명용 날짜 문자열: 20250101_1430 */
function getDateStamp() {
  const d   = new Date();
  const pad = n => String(n).padStart(2, '0');
  return `${d.getFullYear()}${pad(d.getMonth()+1)}${pad(d.getDate())}`
       + `_${pad(d.getHours())}${pad(d.getMinutes())}`;
}

/* ════════════════════════════════════════
   토스트 알림
════════════════════════════════════════ */
let toastTimer = null;

function showToast(msg, color = '#22c55e') {
  const toast = document.getElementById('toast');
  toast.textContent   = msg;
  toast.style.background = color;
  toast.classList.add('show');

  // 이전 타이머가 있으면 초기화 (연속 호출 대비)
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => toast.classList.remove('show'), 2200);
}

/* ════════════════════════════════════════
   이벤트 바인딩
════════════════════════════════════════ */
function bindEvents() {
  // 추가 버튼
  document.getElementById('add-btn').addEventListener('click', () => {
    const input = document.getElementById('movie-input');
    addMovie(input.value);
    input.value = '';
    input.focus();
  });

  // 엔터 키
  document.getElementById('movie-input').addEventListener('keydown', e => {
    if (e.key === 'Enter') {
      const input = e.currentTarget;
      addMovie(input.value);
      input.value = '';
    }
  });

  // JSON 저장 버튼
  document.getElementById('save-btn').addEventListener('click', saveAsJSON);
}

/* ════════════════════════════════════════
   앱 초기화 (진입점)
════════════════════════════════════════ */
function init() {
  loadFromStorage();  // 1. 데이터 로드
  render();           // 2. 화면 그리기
  setupDropZones();   // 3. 드롭존 이벤트 등록
  bindEvents();       // 4. 버튼/입력 이벤트 등록
}

init();
</script>
</body>
</html>
